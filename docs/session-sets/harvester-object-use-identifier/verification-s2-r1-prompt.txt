## Strategy (authoritative spec)

# Object Use Identification Strategy

> **Status:** Authored in Session 1 of `harvester-object-use-identifier`.
> **Authority:** This document is the design specification for the
> `AccessHarvester identify-usage` subcommand. Session 2 implements it
> exactly as described here; divergences require human approval.

---

## 1. Purpose

The `identify-usage` subcommand analyses KB/IR output from a harvested
Access database and produces two artefacts:

1. `object-use.json` — intermediate structured findings per object
   (classification, confidence, fired signals, clarifying questions).
2. `object-use-checklist.json` — a `uat-checklist-editor`-compatible
   JSON checklist the developer and user review together.

All input signals come from already-harvested KB/IR files. No Access COM
is required.

---

## 2. Usage Classification Taxonomy

Seven mutually exclusive categories assigned to each object:

| Category | Description | Typical objects |
|---|---|---|
| **MainDataEntry** | Form backed by a transactional table; VBA handlers are CRUD-focused (BeforeUpdate, AfterUpdate, OnDelete) | `frmClientDetail`, `frmProviderDetail` |
| **LookupReference** | Form or table holding code lists; referenced as RowSource by combo/list controls; few or no FK children | `tlkpStates`, `tlkpPlanStatuses`, combo-only forms |
| **Reporting** | Report or select-only query not used as the record source for a form; presents data without writing it | `rptClientProfile`, `qryClientList` |
| **NavigationShell** | Form with no RecordSource; opens other forms or reports via VBA/macros (DoCmd.OpenForm, OpenReport) | `frmSwitchboard`, `frmReportMenu` |
| **StandaloneUtility** | Table or query not referenced as RecordSource, RowSource, VBA target, macro target, or FK peer; may be used in ad-hoc queries or exports | staging tables, scratch queries |
| **PossiblyUnused** | Object absent from all six signal types; no reference found anywhere in the KB/IR | — |
| **Unknown** | Insufficient signals to classify with confidence ≥ 0.2, or classification is contradictory | form with no RecordSource and no VBA |

**Confidence** is a 0.0–1.0 score:
- ≥ 0.8 = High
- 0.5–0.8 = Medium
- 0.2–0.5 = Low
- < 0.2 = Unknown (category becomes `Unknown` regardless of signals)

using System;
using System.Collections.Generic;
using System.Linq;

namespace AccessHarvester.Pipeline.UsageIdentification;

/// <summary>
/// Classifies all non-excluded objects in an <see cref="IrDatabase"/> using
/// the six signal families and the rule tables defined in
/// <c>docs/planning/object-use-identification-strategy.md</c> §4.
/// </summary>
public static class UsageClassifier
{
    // QueryTypeCode constants per Access DAO
    private const int TypeSelect     = 0;
    private const int TypeCrosstab   = 1;
    private const int TypeDelete     = 2;
    private const int TypeUpdate     = 3;
    private const int TypeAppend     = 4;
    private const int TypeMakeTable  = 5;
    private const int TypeDataDef    = 6;
    private const int TypeUnion      = 64;
    private const int TypePassThrough = 128;

    /// <summary>
    /// Produces one <see cref="UsageFinding"/> per object in the database.
    /// Objects in <paramref name="excludedObjects"/> are skipped.
    /// </summary>
    public static IReadOnlyList<UsageFinding> Classify(
        IrDatabase db,
        SignalIndex index,
        IReadOnlySet<string>? excludedObjects = null)
    {
        var findings = new List<UsageFinding>();

        foreach (var form in db.Forms)
        {
            if (IsExcluded(form.Name, excludedObjects)) continue;
            findings.Add(ClassifyForm(form, index));
        }
        foreach (var report in db.Reports)
        {
            if (IsExcluded(report.Name, excludedObjects)) continue;
            findings.Add(ClassifyReport(report, index));
        }
        foreach (var query in db.Queries)
        {
            if (IsExcluded(query.Name, excludedObjects)) continue;
            findings.Add(ClassifyQuery(query, index));
        }
        foreach (var table in db.Tables)
        {
            if (IsExcluded(table.Name, excludedObjects)) continue;
            if (IsSystemTable(table.Name)) continue;
            findings.Add(ClassifyTable(table, index));
        }

        return findings;
    }

    // ── Forms ────────────────────────────────────────────────────────────────────

    private static UsageFinding ClassifyForm(IrFormOrReport form, SignalIndex index)
    {
        bool hasRecordSource = index.RecordSourceByObject.TryGetValue(form.Name, out string? rs);
        bool hasCrudHandlers = form.HasBeforeUpdate || form.HasAfterUpdate || form.HasOnDelete;
        bool hasNavEdgesOut  = index.NavigationSources.Contains(form.Name);
        bool hasVbaCode      = !string.IsNullOrWhiteSpace(form.VbaCode);

        string category;
        double confidence;
        var evidence = new List<string>();

        if (hasRecordSource && rs != null)
        {
            evidence.Add($"RecordSource={rs}");
            bool rsIsTransactional = index.HasFkChildren(rs);
            bool rsIsLookup = index.IsLookupTable(rs) || IsLookupSource(rs, index);

            if (rsIsTransactional && hasCrudHandlers)
            {
                // F1
                category   = UsageCategory.MainDataEntry;
                confidence = 0.85;
                if (hasCrudHandlers) evidence.Add("CRUD-VBA-handlers");
            }
            else if (rsIsTransactional)
            {
                // F2
                category   = UsageCategory.MainDataEntry;
                confidence = 0.65;
            }
            else if (rsIsLookup)
            {
                // F3
                category   = UsageCategory.LookupReference;
                confidence = 0.75;
            }
            else
            {
                // F4 — RecordSource is a query; sub-classify by query type
                var q = FindQuery(rs, index);
                if (q != null && IsActionQueryType(q.QueryTypeCode))
                {
                    category   = UsageCategory.MainDataEntry;
                    confidence = 0.55;
                }
                else
                {
                    category   = UsageCategory.Reporting;
                    confidence = 0.55;
                }
            }
        }
        else if (hasNavEdgesOut)
        {
            // F5
            category   = UsageCategory.NavigationShell;
            confidence = 0.80;
            evidence.Add("has-navigation-edges-out");
        }
        else if (hasVbaCode)
        {
            // F6
            category   = UsageCategory.Unknown;
            confidence = 0.30;
            evidence.Add("has-VBA-no-record-source");
        }
        else
        {
            // F7
            category   = UsageCategory.PossiblyUnused;
            confidence = 0.70;
        }

        confidence = AdjustConfidence(confidence, form.Name, "Form", index,
                                      hasRecordSource, hasNavEdgesOut, hasVbaCode, 0);
        if (confidence < 0.2) category = UsageCategory.Unknown;

        return BuildFinding("Form", form.Name, category, confidence, evidence,
                            "Form", form.RecordSource, hasRecordSource,


## QuestionGenerator.cs

using System.Collections.Generic;
using System.Linq;

namespace AccessHarvester.Pipeline.UsageIdentification;

/// <summary>
/// Generates clarifying questions for a classified object based on the
/// uncertainty signals defined in strategy §5-6.
/// </summary>
public static class QuestionGenerator
{
    /// <summary>
    /// Returns the set of clarifying questions for this object given its
    /// classification context and signal entries.
    /// </summary>
    public static IReadOnlyList<ClarifyingQuestion> Generate(
        string objectType, string objectName,
        string category, double confidence,
        IReadOnlyList<SignalEntry> signals,
        string? recordSource, bool hasVbaCode,
        bool isNavigationTarget, int? queryTypeCode,
        SignalIndex index)
    {
        var questions = new List<ClarifyingQuestion>();

        // NoNavigationEntry — form not present as navigation target and not a nav shell
        if (objectType == "Form"
            && !isNavigationTarget
            && category != UsageCategory.NavigationShell
            && category != UsageCategory.PossiblyUnused)
        {
            questions.Add(new ClarifyingQuestion
            {
                QuestionText    = $"No navigation entry found for Form '{objectName}' — is it launched from VBA code, another form, or an external application?",
                UncertaintySignal = UncertaintySignal.NoNavigationEntry,
            });
        }

        // NoRecordSource — form has no record source and is not NavigationShell or PossiblyUnused
        if (objectType == "Form" && string.IsNullOrWhiteSpace(recordSource)
            && category is not (UsageCategory.NavigationShell or UsageCategory.PossiblyUnused))
        {
            questions.Add(new ClarifyingQuestion
            {
                QuestionText    = $"Form '{objectName}' has no record source — is it a navigation panel, a parameter/filter dialog, or a utility form?",
                UncertaintySignal = UncertaintySignal.NoRecordSource,
            });
        }

        // QueryReferencesNoTables — query SQL references only other queries
        if (objectType == "Query")
        {
            bool noDirectTables = signals
                .FirstOrDefault(s => s.SignalType == UsageSignalType.QueryTableReference)?.Evidence
                .Any(e => e.Contains("references-no-tables")) ?? false;
            if (noDirectTables)
            {
                questions.Add(new ClarifyingQuestion
                {
                    QuestionText    = $"Query '{objectName}' references only other queries — is it an intermediate computation step, a parameter source, or can it be merged into its callers?",
                    UncertaintySignal = UncertaintySignal.QueryReferencesNoTables,
                });
            }
        }

        // CrosstabQuery — query is crosstab type
        if (objectType == "Query" && queryTypeCode == 1)
        {
            questions.Add(new ClarifyingQuestion
            {
                QuestionText    = $"Query '{objectName}' is a crosstab query — is it used for reporting, for driving a form or chart control, or for data exports?",
                UncertaintySignal = UncertaintySignal.CrosstabQuery,
            });
        }

        // UnreferencedObject — object is completely absent from all signals
        if (category == UsageCategory.PossiblyUnused)
        {
            questions.Add(new ClarifyingQuestion
            {
                QuestionText    = $"{objectType} '{objectName}' is not referenced by any form, report, query, VBA, or macro — is it still in use or can it be excluded from analysis?",
                UncertaintySignal = UncertaintySignal.UnreferencedObject,
            });
        }

        // LookupTablePattern — table classified as LookupReference
        if (objectType == "Table" && category == UsageCategory.LookupReference)
        {
            questions.Add(new ClarifyingQuestion
            {
                QuestionText    = $"Table '{objectName}' appears to hold reference/lookup values — should it migrate as static reference data, or be replaced with application constants?",
                UncertaintySignal = UncertaintySignal.LookupTablePattern,
            });
        }

        // UnrecognizedVbaPattern — form/module has VBA but no parseable DoCmd/DFunction calls
        if (hasVbaCode && category == UsageCategory.Unknown
            && objectType is "Form" or "Module")
        {
            questions.Add(new ClarifyingQuestion
            {
                QuestionText    = $"{objectType} '{objectName}' contains VBA that couldn't be auto-classified — can you describe its primary function?",
                UncertaintySignal = UncertaintySignal.UnrecognizedVbaPattern,
            });
        }

        // CrypticObjectName — object name has no conventional prefix and < 5 chars
        if (IsCrypticName(objectName))
        {
            questions.Add(new ClarifyingQuestion
            {
                QuestionText    = $"{objectType} '{objectName}' has a name that doesn't clearly indicate its purpose — can you describe what it is used for?",
                UncertaintySignal = UncertaintySignal.CrypticObjectName,
            });
        }

        return questions;
    }

    private static bool IsCrypticName(string name)
        => name.Length < 5 && !name.Contains('_') && !char.IsUpper(name.Length > 1 ? name[1] : '\0');
}

using System;
using System.Collections.Generic;
using System.Linq;

namespace AccessHarvester.Pipeline.UsageIdentification;

/// <summary>
/// Cross-reference index built from all signal collectors over an <see cref="IrDatabase"/>.
/// Provides O(1) lookup for the classifier's rules.
/// </summary>
public sealed class SignalIndex
{
    // RecordSource: formOrReportName -> tableName/queryName
    public IReadOnlyDictionary<string, string> RecordSourceByObject { get; }

    // Objects used AS a RecordSource (by any form/report)
    public IReadOnlySet<string> UsedAsRecordSource { get; }

    // Objects used AS a RecordSource specifically by a form
    public IReadOnlySet<string> UsedAsFormRecordSource { get; }

    // Objects used AS a RecordSource specifically by a report
    public IReadOnlySet<string> UsedAsReportRecordSource { get; }

    // Objects used AS a RowSource (by any form control or VBA assignment)
    public IReadOnlySet<string> UsedAsRowSource { get; }

    // Forms/objects that ARE navigation targets (opened by another object)
    public IReadOnlySet<string> NavigationTargets { get; }

    // Forms/objects that emit navigation edges (open other objects)
    public IReadOnlySet<string> NavigationSources { get; }

    // Tables/queries referenced in VBA domain functions or OpenRecordset
    public IReadOnlySet<string> VbaReferencedObjects { get; }

    // Query -> set of tables/queries it references in FROM/JOIN
    public IReadOnlyDictionary<string, IReadOnlySet<string>> QuerySqlTargetsByQuery { get; }

    // Tables/queries that appear in any query's SQL
    public IReadOnlySet<string> UsedInQuerySql { get; }

    // FK: parent table -> set of child tables
    public IReadOnlyDictionary<string, IReadOnlySet<string>> FkChildrenByTable { get; }

    // FK: child table -> set of parent tables
    public IReadOnlyDictionary<string, IReadOnlySet<string>> FkParentsByTable { get; }

    // Tables whose name matches the lookup naming convention
    public IReadOnlySet<string> LookupNamedTables { get; }

    private SignalIndex(
        IReadOnlyDictionary<string, string> recordSourceByObject,
        IReadOnlySet<string> usedAsRecordSource,
        IReadOnlySet<string> usedAsFormRecordSource,
        IReadOnlySet<string> usedAsReportRecordSource,
        IReadOnlySet<string> usedAsRowSource,
        IReadOnlySet<string> navigationTargets,
        IReadOnlySet<string> navigationSources,
        IReadOnlySet<string> vbaReferencedObjects,
        IReadOnlyDictionary<string, IReadOnlySet<string>> querySqlTargetsByQuery,
        IReadOnlySet<string> usedInQuerySql,
        IReadOnlyDictionary<string, IReadOnlySet<string>> fkChildrenByTable,
        IReadOnlyDictionary<string, IReadOnlySet<string>> fkParentsByTable,
        IReadOnlySet<string> lookupNamedTables)
    {
        RecordSourceByObject    = recordSourceByObject;
        UsedAsRecordSource      = usedAsRecordSource;
        UsedAsFormRecordSource  = usedAsFormRecordSource;
        UsedAsReportRecordSource = usedAsReportRecordSource;
        UsedAsRowSource         = usedAsRowSource;
        NavigationTargets       = navigationTargets;
        NavigationSources       = navigationSources;
        VbaReferencedObjects    = vbaReferencedObjects;
        QuerySqlTargetsByQuery  = querySqlTargetsByQuery;
        UsedInQuerySql          = usedInQuerySql;
        FkChildrenByTable       = fkChildrenByTable;
        FkParentsByTable        = fkParentsByTable;
        LookupNamedTables       = lookupNamedTables;
    }

    /// <summary>Builds a <see cref="SignalIndex"/> from an <see cref="IrDatabase"/>.</summary>
    public static SignalIndex Build(IrDatabase db)
    {
        // RecordSources — split into form-level and report-level for Q2/Q3 classification
        var recordSourceByObject = RecordSourceCollector.Collect(db);
        var usedAsFormRecordSource = new HashSet<string>(
            db.Forms
              .Where(f => f.RecordSource != null && !f.RecordSource.TrimStart().StartsWith("SELECT", StringComparison.OrdinalIgnoreCase))
              .Select(f => f.RecordSource!.Trim()),
            StringComparer.OrdinalIgnoreCase);
        var usedAsReportRecordSource = new HashSet<string>(
            db.Reports
              .Where(r => r.RecordSource != null && !r.RecordSource.TrimStart().StartsWith("SELECT", StringComparison.OrdinalIgnoreCase))
              .Select(r => r.RecordSource!.Trim()),
            StringComparer.OrdinalIgnoreCase);
        var usedAsRecordSource = new HashSet<string>(
            recordSourceByObject.Values, StringComparer.OrdinalIgnoreCase);

        // RowSources
        var rowSourceTriples = RowSourceCollector.Collect(db);
        var usedAsRowSource = new HashSet<string>(
            rowSourceTriples.Select(t => t.RowSourceName), StringComparer.OrdinalIgnoreCase);

        // Navigation edges
        var navEdges = NavigationEdgeCollector.Collect(db);
        var navigationTargets = new HashSet<string>(
            navEdges.Select(e => e.Target), StringComparer.OrdinalIgnoreCase);
        var navigationSources = new HashSet<string>(
            navEdges.Select(e => e.Source), StringComparer.OrdinalIgnoreCase);

        // VBA references
        var vbaRefs = VbaReferenceCollector.Collect(db);
        var vbaReferencedObjects = new HashSet<string>(
            vbaRefs.Select(r => r.ReferencedObject), StringComparer.OrdinalIgnoreCase);

        // Query SQL references
        var queryRefs = QueryTableReferenceCollector.Collect(db);
        var querySqlTargetsByQuery = queryRefs
            .GroupBy(r => r.QueryName, StringComparer.OrdinalIgnoreCase)
            .ToDictionary(
                g => g.Key,
                g => (IReadOnlySet<string>)new HashSet<string>(g.Select(r => r.ReferencedObject),
                                                                StringComparer.OrdinalIgnoreCase),
                StringComparer.OrdinalIgnoreCase);
        var usedInQuerySql = new HashSet<string>(
            queryRefs.Select(r => r.ReferencedObject), StringComparer.OrdinalIgnoreCase);

        // FK relationships
        var fkEdges = RelationshipCollector.Collect(db);
        var fkChildrenByTable = fkEdges
            .GroupBy(e => e.ParentTable, StringComparer.OrdinalIgnoreCase)
            .ToDictionary(
                g => g.Key,
                g => (IReadOnlySet<string>)new HashSet<string>(g.Select(e => e.ChildTable),
                                                                StringComparer.OrdinalIgnoreCase),
                StringComparer.OrdinalIgnoreCase);
        var fkParentsByTable = fkEdges
            .GroupBy(e => e.ChildTable, StringComparer.OrdinalIgnoreCase)
            .ToDictionary(
                g => g.Key,
                g => (IReadOnlySet<string>)new HashSet<string>(g.Select(e => e.ParentTable),
                                                                StringComparer.OrdinalIgnoreCase),
                StringComparer.OrdinalIgnoreCase);

        // Lookup naming convention
        var lookupNamedTables = new HashSet<string>(
            db.Tables
              .Where(t => IsLookupNamed(t.Name))
              .Select(t => t.Name),
            StringComparer.OrdinalIgnoreCase);

        return new SignalIndex(
            recordSourceByObject,
            usedAsRecordSource,
            usedAsFormRecordSource,
            usedAsReportRecordSource,
            usedAsRowSource,
            navigationTargets,
            navigationSources,
            vbaReferencedObjects,
            querySqlTargetsByQuery,
            usedInQuerySql,
            fkChildrenByTable,
            fkParentsByTable,
            lookupNamedTables);
    }

    // True when a table name matches the conventional lookup/reference naming prefixes.
    private static bool IsLookupNamed(string name)
        => name.StartsWith("tlkp", StringComparison.OrdinalIgnoreCase)
        || name.StartsWith("lkp",  StringComparison.OrdinalIgnoreCase)
        || name.Contains("_Lookup", StringComparison.OrdinalIgnoreCase)
        || name.Contains("_Codes",  StringComparison.OrdinalIgnoreCase)
        || name.Contains("_Types",  StringComparison.OrdinalIgnoreCase);

    /// <summary>Returns true when the table has FK children in the index.</summary>
    public bool HasFkChildren(string tableName)
        => FkChildrenByTable.ContainsKey(tableName);

    /// <summary>Returns true when the table has FK parents in the index.</summary>
    public bool HasFkParents(string tableName)
        => FkParentsByTable.ContainsKey(tableName);

    /// <summary>Returns true when the table is a lookup-named table.</summary>
    public bool IsLookupTable(string tableName)
        => LookupNamedTables.Contains(tableName);

    /// <summary>Returns true when no signal references this object anywhere.</summary>
    public bool IsCompletelyUnreferenced(string objectName)
        => !UsedAsRecordSource.Contains(objectName)
        && !UsedAsRowSource.Contains(objectName)
        && !NavigationTargets.Contains(objectName)
        && !VbaReferencedObjects.Contains(objectName)
        && !UsedInQuerySql.Contains(objectName)
        && !HasFkChildren(objectName)
        && !HasFkParents(objectName);
}

    private static void WriteObjectUse(ObjectUseFindings findings, string path)

            findings.Database,
            findings.GeneratedAt,
            findings.ExcludedObjects,
            findings.Objects.Select(ToJsonFinding).ToArray());

    private static ObjectUseFindingJson ToJsonFinding(UsageFinding f)

            var steps = f.Questions.Select((q, i) => new ChecklistStep(
                StepNumber: i + 1,
                Action: q.QuestionText,
                Verifications: []
            )).ToArray();

                FunctionalArea: f.ObjectType,
                Subarea: f.ObjectName,
                ItemLabel: f.UsageSummary,
                Steps: steps,
                IsJudgmentItem: true,
                Passes: false,
                Feedback: ""
            ));

            ProjectName: $"{findings.Database} — Object Use Checklist",
            Review: reviews.ToArray());
    }

        var doc = BuildChecklist(findings);
        string json = JsonSerializer.Serialize(doc, _jsonOpts);

    }

## NavigationEdgeCollector.cs

using System;
using System.Collections.Generic;
using System.Text.RegularExpressions;

namespace AccessHarvester.Pipeline.UsageIdentification;

/// <summary>
/// Collects navigation edges from macro IR files and VBA code.
/// A navigation edge is a directed (source, target, edgeType) triple where
/// the source object opens the target via DoCmd.OpenForm / DoCmd.OpenReport
/// or the equivalent macro action.
/// </summary>
public static class NavigationEdgeCollector
{
    // DoCmd.OpenForm "FormName" or DoCmd.OpenReport "ReportName"
    private static readonly Regex _doCmdPattern =
        new(@"DoCmd\.Open(?<verb>Form|Report|Query)\s*\(?\s*""(?<name>[^""]+)""",
            RegexOptions.IgnoreCase | RegexOptions.Compiled);

    // basListHelpers.OpenDetailFromList ..., "FormName" — common helper in HomeHealthCare
    private static readonly Regex _helperOpenPattern =
        new(@"OpenDetailFromList\s+\w+\s*,\s*""(?<name>[^""]+)""",
            RegexOptions.IgnoreCase | RegexOptions.Compiled);

    public sealed record NavigationEdge(string Source, string Target, string EdgeType);

    /// <summary>
    /// Returns all directed navigation edges found in macros and VBA code.
    /// </summary>
    public static IReadOnlyList<NavigationEdge> Collect(IrDatabase db)
    {
        var edges = new List<NavigationEdge>();

        // Macro-level navigation edges
        foreach (var macro in db.Macros)
            AddMacroEdges(macro.Name, macro.Actions, edges);

        // Form-level VBA navigation edges (form code-behind)
        foreach (var form in db.Forms)
        {
            if (form.VbaCode is not null)
                AddVbaEdges(form.Name, form.VbaCode, edges);
            // Embedded macro action nodes in the form's Children
            AddMacroEdges(form.Name, form.MacroActions, edges);
        }

        // Report-level VBA
        foreach (var report in db.Reports)
            if (report.VbaCode is not null)
                AddVbaEdges(report.Name, report.VbaCode, edges);

        // Standalone module VBA
        foreach (var module in db.Modules)
            if (module.VbaCode is not null)
                AddVbaEdges(module.Name, module.VbaCode, edges);

        return edges;
    }

    private static void AddMacroEdges(string source, IReadOnlyList<IrMacroAction> actions,
                                      List<NavigationEdge> edges)
    {
        foreach (var action in actions)
        {
            string? edgeType = action.Action switch
            {
                var a when string.Equals(a, "OpenForm",   StringComparison.OrdinalIgnoreCase) => "OpenForm",
                var a when string.Equals(a, "OpenReport", StringComparison.OrdinalIgnoreCase) => "OpenReport",
                var a when string.Equals(a, "OpenQuery",  StringComparison.OrdinalIgnoreCase) => "OpenQuery",
                _ => null
            };
            if (edgeType == null) continue;
            string? target = action.Arguments.Count > 0 ? action.Arguments[0] : null;
            if (!string.IsNullOrWhiteSpace(target))
                edges.Add(new NavigationEdge(source, target.Trim(), edgeType));
        }
    }

    private static void AddVbaEdges(string source, string vbaCode, List<NavigationEdge> edges)
    {
        foreach (Match m in _doCmdPattern.Matches(vbaCode))
        {
            string verb   = m.Groups["verb"].Value;
            string target = m.Groups["name"].Value.Trim();
            if (!string.IsNullOrEmpty(target))
                edges.Add(new NavigationEdge(source, target, "Open" + verb));
        }
        foreach (Match m in _helperOpenPattern.Matches(vbaCode))
        {
            string target = m.Groups["name"].Value.Trim();
            if (!string.IsNullOrEmpty(target))
                edges.Add(new NavigationEdge(source, target, "OpenForm"));
        }
    }
}


## RecordSourceCollector.cs

using System;
using System.Collections.Generic;

namespace AccessHarvester.Pipeline.UsageIdentification;

/// <summary>
/// Collects RecordSource bindings from form and report IR files.
/// Produces a map from object name (form or report) to its RecordSource
/// (a table or query name). Only non-SQL RecordSources are retained;
/// inline SQL strings (starting with SELECT) are omitted.
/// </summary>
public static class RecordSourceCollector
{
    /// <summary>
    /// Returns a dictionary mapping each form/report name to its RecordSource.
    /// Objects with no RecordSource or with an inline SQL RecordSource are absent.
    /// </summary>
    public static IReadOnlyDictionary<string, string> Collect(IrDatabase db)
    {
        var map = new Dictionary<string, string>(StringComparer.OrdinalIgnoreCase);
        foreach (var form in db.Forms)
            AddIfNamedSource(map, form.Name, form.RecordSource);
        foreach (var report in db.Reports)
            AddIfNamedSource(map, report.Name, report.RecordSource);
        return map;
    }

    private static void AddIfNamedSource(Dictionary<string, string> map, string name, string? source)
    {
        if (string.IsNullOrWhiteSpace(source)) return;
        string trimmed = source.Trim();
        // Exclude inline SQL strings — they start with SELECT (case-insensitive)
        if (trimmed.StartsWith("SELECT", StringComparison.OrdinalIgnoreCase)) return;
        map[name] = trimmed;
    }
}


## VbaReferenceCollector.cs

using System;
using System.Collections.Generic;
using System.Text.RegularExpressions;

namespace AccessHarvester.Pipeline.UsageIdentification;

/// <summary>
/// Collects table/query references found in VBA code via domain aggregate
/// functions (DCount, DLookup, DSum, DMin, DMax, DFirst, DLast, DAvg) and
/// CurrentDb().OpenRecordset calls. These patterns reveal tables that are
/// used programmatically without a formal RecordSource binding.
/// </summary>
public static class VbaReferenceCollector
{
    // DCount/DLookup/etc. with a quoted table/query name as the second argument
    private static readonly Regex _domainFunctionPattern =
        new(@"\bD(?:Count|Lookup|Sum|Min|Max|First|Last|Avg)\s*\(\s*[^,]+,\s*""([^""]+)""",
            RegexOptions.IgnoreCase | RegexOptions.Compiled);

    // CurrentDb.OpenRecordset("tableName") or OpenRecordset("tableName")
    private static readonly Regex _openRecordsetPattern =
        new(@"OpenRecordset\s*\(\s*""([^""]+)""",
            RegexOptions.IgnoreCase | RegexOptions.Compiled);

    public sealed record VbaReference(string SourceObject, string ReferencedObject);

    /// <summary>
    /// Returns all VBA-level references to table/query names discovered in
    /// form, report, and module VBA code.
    /// </summary>
    public static IReadOnlyList<VbaReference> Collect(IrDatabase db)
    {
        var results = new List<VbaReference>();
        foreach (var form in db.Forms)
            if (form.VbaCode is not null)
                ScanVba(form.Name, form.VbaCode, results);
        foreach (var report in db.Reports)
            if (report.VbaCode is not null)
                ScanVba(report.Name, report.VbaCode, results);
        foreach (var module in db.Modules)
            if (module.VbaCode is not null)
                ScanVba(module.Name, module.VbaCode, results);
        return results;
    }

    private static void ScanVba(string sourceObject, string vbaCode, List<VbaReference> results)
    {
        foreach (Match m in _domainFunctionPattern.Matches(vbaCode))
        {
            string name = m.Groups[1].Value.Trim();
            if (!string.IsNullOrEmpty(name))
                results.Add(new VbaReference(sourceObject, name));
        }
        foreach (Match m in _openRecordsetPattern.Matches(vbaCode))
        {
            string name = m.Groups[1].Value.Trim();
            // Exclude inline SQL
            if (!string.IsNullOrEmpty(name)
                && !name.StartsWith("SELECT", StringComparison.OrdinalIgnoreCase))
                results.Add(new VbaReference(sourceObject, name));
        }
    }
}
