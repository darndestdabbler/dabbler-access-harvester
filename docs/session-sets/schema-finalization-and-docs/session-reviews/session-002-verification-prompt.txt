### AccessHarvester.Pipeline/Validation/SchemaValidator.cs
```csharp
using System;
using System.Collections.Concurrent;
using System.Collections.Generic;
using System.IO;
using System.Text.Json;
using Json.Schema;

namespace AccessHarvester.Pipeline.Validation;

/// <summary>
/// Validates harvester output documents against harvester-schema.v1.0.json.
/// Hard-fails the caller on schema non-conformance; conformant documents pass silently.
/// </summary>
public sealed class SchemaValidator
{
    // JsonSchema.FromText registers the schema globally by its $id on first load.
    // Cache by absolute path to avoid duplicate-registration errors when multiple
    // instances are created with the same schema file.
    private static readonly ConcurrentDictionary<string, JsonSchema> _cache = new(StringComparer.OrdinalIgnoreCase);

    private readonly JsonSchema _schema;

    public SchemaValidator(string schemaJsonPath)
    {
        ArgumentException.ThrowIfNullOrEmpty(schemaJsonPath);
        string fullPath = Path.GetFullPath(schemaJsonPath);
        _schema = _cache.GetOrAdd(fullPath, path =>
        {
            var schemaText = File.ReadAllText(path, System.Text.Encoding.UTF8);
            return JsonSchema.FromText(schemaText);
        });
    }

    /// <summary>
    /// Validates a JSON document string against the loaded schema.
    /// Returns a result containing validity and any error messages.
    /// </summary>
    public SchemaValidationResult Validate(string documentJson)
    {
        ArgumentException.ThrowIfNullOrEmpty(documentJson);

        using var document = JsonDocument.Parse(documentJson);
        var options = new EvaluationOptions { OutputFormat = OutputFormat.List };
        EvaluationResults results = _schema.Evaluate(document.RootElement, options);

        if (results.IsValid)
            return SchemaValidationResult.Valid();

        var errors = new List<string>();
        foreach (var detail in results.Details)
        {
            if (!detail.IsValid && detail.Errors is not null)
            {
                foreach (var (keyword, message) in detail.Errors)
                    errors.Add($"{detail.InstanceLocation}: [{keyword}] {message}");
            }
        }

        return SchemaValidationResult.Invalid(errors);
    }
}
```

### AccessHarvester.Pipeline/Validation/SchemaValidationResult.cs
```csharp
using System;
using System.Collections.Generic;

namespace AccessHarvester.Pipeline.Validation;

/// <summary>Result of validating a document against the harvester schema.</summary>
public sealed class SchemaValidationResult
{
    public bool IsValid { get; }
    public IReadOnlyList<string> Errors { get; }

    private SchemaValidationResult(bool isValid, IReadOnlyList<string> errors)
    {
        IsValid = isValid;
        Errors = errors;
    }

    public static SchemaValidationResult Valid() =>
        new(true, Array.Empty<string>());

    public static SchemaValidationResult Invalid(IReadOnlyList<string> errors) =>
        new(false, errors);
}
```

### tests/AccessHarvester.Pipeline.Tests/Validation/SchemaValidatorTests.cs
```csharp
using System;
using System.IO;
using AccessHarvester.Pipeline.Validation;
using Xunit;

namespace AccessHarvester.Pipeline.Tests.Validation;

public sealed class SchemaValidatorTests
{
    private static string LocateSchemaPath()
    {
        string dir = AppContext.BaseDirectory;
        for (int i = 0; i < 12; i++)
        {
            string candidate = Path.Combine(dir, "docs", "test-cases", "harvester-schema.v1.0.json");
            if (File.Exists(candidate)) return candidate;
            string? parent = Directory.GetParent(dir)?.FullName;
            if (parent is null || parent == dir) break;
            dir = parent;
        }
        throw new FileNotFoundException(
            $"Could not locate harvester-schema.v1.0.json walking up from '{AppContext.BaseDirectory}'.");
    }

    private static readonly string SchemaPath = LocateSchemaPath();

    private static SchemaValidator CreateValidator() => new(SchemaPath);

    // (a) A conformant minimal DatabaseManifest returns no errors.
    [Fact]
    public void Validate_ConformantDocument_ReturnsValid()
    {
        const string json = """
            {
              "SchemaType": "DatabaseManifest",
              "SchemaVersion": "1.0",
              "HarvesterVersion": "1.0.0",
              "Database": { "Name": "Test.accdb" }
            }
            """;

        var result = CreateValidator().Validate(json);

        Assert.True(result.IsValid, $"Expected VALID but got errors: {string.Join("; ", result.Errors)}");
        Assert.Empty(result.Errors);
    }

    // (b) A document missing a required field (SchemaType) returns errors.
    [Fact]
    public void Validate_MissingRequiredField_ReturnsInvalid()
    {
        const string json = """
            {
              "SchemaVersion": "1.0",
              "HarvesterVersion": "1.0.0",
              "Database": { "Name": "Test.accdb" }
            }
            """;

        var result = CreateValidator().Validate(json);

        Assert.False(result.IsValid, "Expected INVALID (missing SchemaType) but got VALID.");
        Assert.NotEmpty(result.Errors);
    }

    // (c) A document with an unknown extra field is accepted (additionalProperties not forbidden at v1).
    [Fact]
    public void Validate_UnknownField_IsAccepted()
    {
        const string json = """
            {
              "SchemaType": "DatabaseManifest",
              "SchemaVersion": "1.0",
              "HarvesterVersion": "1.0.0",
              "Database": { "Name": "Test.accdb" },
              "UnknownFutureField": "some-value"
            }
            """;

        var result = CreateValidator().Validate(json);

        Assert.True(result.IsValid, $"Expected VALID (unknown fields allowed) but got errors: {string.Join("; ", result.Errors)}");
    }
}
```
