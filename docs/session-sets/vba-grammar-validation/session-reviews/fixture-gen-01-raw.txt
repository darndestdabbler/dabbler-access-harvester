```vba
'--------------------------------------------------------------------------------
' H-D05: Resume / Resume Next / Resume <label>
'
' Hazard: Tests all four Resume forms per MS-VBAL §5.4.12.
' 1. Resume (retry the failing statement)
' 2. Resume 0 (equivalent to Resume)
' 3. Resume Next (continue at statement after the failing statement)
' 4. Resume <label> (jump to a named label, also clears Err object)
' The disambiguation between 'Resume 0' and 'GoTo 0' is the key test.
'--------------------------------------------------------------------------------
Option Explicit

Public Sub ResumeForms(resumeType As Integer)
    On Error GoTo ErrorHandler

    ' A statement that will error
    Debug.Print 1 / 0

ResumeTarget:
    Exit Sub

ErrorHandler:
    Select Case resumeType
        Case 1
            Resume
        Case 2
            Resume 0
        Case 3
            Resume Next
        Case 4
            Resume ResumeTarget
    End Select
End Sub
```
