'--------------------------------------------------------------------------------
' Module: Module_OnErrorHazard
' Source: 05-h-d04-on-error-goto-zero.bas
' Hazard: H-D04 (On Error GoTo 0)
'
' Purpose:
' Demonstrates the 'On Error GoTo 0' statement which deactivates the
' current error handler. Also includes the other 'On Error' variants
' for completeness.
'--------------------------------------------------------------------------------
Option Explicit

Public Sub DemonstrateOnErrorVariants()
    ' 1. Activate a handler
    On Error GoTo ErrHandler
    
    ' 2. Error-prone code (for context)
    Dim x As Integer
    x = 1 / 0
    
    ' 3. Deactivate the handler
    On Error GoTo 0
    
    ' 4. Code outside error-handling scope
    x = 1
    
    ' Also show the third variant
    On Error Resume Next
    
Exit Sub ' Prevent fall-through into handler

ErrHandler:
    ' 5. The handler label with a Resume statement
    Resume Next
End Sub
