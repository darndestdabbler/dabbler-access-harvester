' H-D12: Named arguments at call sites (:=)
'
' This module tests the parsing of named arguments in procedure calls.
' It includes:
' - A procedure with Optional parameters and a ParamArray.
' - A call site using only named arguments.
' - A call site mixing positional and named arguments.
' - A call to an external library (DoCmd) using named arguments.

Option Explicit

' Define constant used in DoCmd call for self-containment
Public Const acFormAdd As Long = 2

Public Sub ProcedureWithNamedArgs(Positional1 As String, Optional Opt1 As Long = 1, Optional Opt2 As String = "B", ParamArray OtherArgs() As Variant)
    ' This procedure exists to be a valid call target.
End Sub

Public Sub TestNamedArgCalls()
    ' 1. Pure named argument call, skipping an optional parameter.
    ProcedureWithNamedArgs Opt2:="Skipped Opt1", Positional1:="Value1"

    ' 2. Mixed positional and named arguments.
    ProcedureWithNamedArgs "Value2", Opt2:="Skipped Opt1 again"

    ' 3. Mixed positional, named, and ParamArray arguments.
    ProcedureWithNamedArgs "Value3", Opt2:="C", 100, 200, "end"

    ' 4. Call to external library member with named arguments.
    DoCmd.OpenForm FormName:="frmX", DataMode:=acFormAdd
End Sub
